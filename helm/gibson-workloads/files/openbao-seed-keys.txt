# The OpenBao seed contract (deploy#1732). ONE definition.
#
# The openbao-auto-init sidecar reads this table and makes sure every key
# exists in the kv v2 mount `secret` at `secret/data/<key>` with every
# property listed. Nothing in this repository cross-checks the table against
# the ExternalSecrets that read it. zeroroot-ai/hosted secret-contract.yaml is
# the cross-repo record of which key each consumer reads, and
# zeroroot-ai/hosted scripts/check-secret-contract.py gates it.
#
# Format: one key per line, `<key> <property>:<kind>[:<arg>] ... [@<hours>h]`.
# The optional last field is the lifetime of the key (ADR-0171). When the
# current version is older than it, the sidecar writes a new value for each
# generated property (rand, randb64, rand8, pw). Give a key a lifetime only
# when each consumer has a rotation path that keeps service.
# Comment lines start with `#`.
#
# Kinds, and what the sidecar does when the property is ABSENT:
#   rand           32 random bytes, URL-safe base64, no padding (43 chars)
#   randb64        32 random bytes, standard base64 (44 chars)
#   rand8          8 random bytes, URL-safe base64, no padding (11 chars)
#   pw             43 random letters and digits (256 bits): a PASSWORD, the
#                  kind for every value a program may receive as an argument.
#                  rand starts with '-' one draw in 64, and `neo4j-admin dbms
#                  set-initial-password -Xyz` reads that as an option (exit
#                  64, blocker 3 loops #16/#18). scripts/check-seed-passwords.sh
#                  refuses password:rand.
#   rsapem:<name>  the private key of transit key <name> (rsa-2048), PEM
#   rsapub:<name>  the public key of transit key <name> (rsa-2048), PEM
#   literal:<v>    the literal value <v>
#   empty          the empty string. The operator fills it in later with
#                  scripts/baseline-set-secret.sh
#   input:<file>   the bringup keyring input mounted at /etc/openbao/inputs/<file>
#   inputjson:<f>  the property of the same name in the JSON object mounted at
#                  /etc/openbao/inputs/<f>
#   inputor:<f>:<v> the keyring input <f> when the bringup supplied one, else
#                  the literal <v> (charts#486). A test-mode placeholder stays
#                  only while the keyring member is empty.
#   svctoken       the platform service token the sidecar minted
#   prev:<prop>    the value <prop> had before the last rotation of the key
#                  (ADR-0171), empty before the first one. A consumer that
#                  accepts the old and the new value reads both.
#
# A property that is PRESENT is never overwritten, with two exceptions:
# `input`, `inputjson` and `inputor` follow the keyring (a non-empty keyring value that
# differs from the stored one is written, at sidecar start and on every
# five-minute pass after it, so a member set after the bringup lands without
# a pod restart), and `svctoken` follows the token in the
# gibson-platform-operator-vault Secret. Everything else is generated once
# and kept, so an upgrade never rotates a credential under a workload.
gibson-gibson-platform-postgres-password password:pw
gibson-tenant-admin-postgres-password password:pw
gibson-zitadel-postgres-password password:pw
gibson-openfga-postgres-password password:pw
gibson-spire-postgres-password password:pw
gibson-redis-password password:pw @720h
zitadel-masterkey value:rand
gibson-dashboard-session-secrets auth_secret:rand auth_secret_previous:prev:auth_secret server_actions_key:randb64 @720h
gibson-cg-signing-key key:rand kid:rand8
# The verify-only slots of the CG signing key set. Empty in steady state; the
# sidecar fills and clears them during a rotation (cg_key_rotate, ADR-0171).
gibson-cg-signing-key-next kid:empty key:empty
gibson-cg-signing-key-previous kid:empty key:empty
gibson-zitadel-system-bot-key private_key:rsapem:zitadel-system-bot public_key:rsapub:zitadel-system-bot
# The System API key has two slots, a (above) and b, one System API user each.
# The platform-operator signs with the active slot. The sidecar flips the
# active slot when its key is older than its lifetime, then gives the other
# slot a new key (sys_key_rotate, ADR-0171). The active line comes before
# slot b, so the first seed pass does not read slot b as older than a flip.
gibson-zitadel-system-active slot:literal:a
gibson-zitadel-system-bot-key-b private_key:rsapem:zitadel-system-bot-b public_key:rsapub:zitadel-system-bot-b
ghcr-pull-secret pat:input:ghcr-pull-token
gibson-dashboard-auth-hardening captcha_secret_key:empty email_resend_api_key:empty
gibson-stripe-credentials publishable_key:inputor:stripe-publishable-key:pk_test_placeholder secret_key:inputor:stripe-secret-key:sk_test_placeholder webhook_secret:inputor:stripe-webhook-secret:whsec_placeholder
ses-smtp-credentials username:input:smtp-username password:input:smtp-password
gibson-vault-admin-token token:svctoken
# The monthly drill writes requested_at (a Unix time) to rotate each service
# token now. 0 asks for nothing (ADR-0171).
gibson-service-token-rotation requested_at:literal:0
setec-disk-signing seed:input:setec-disk-signing-seed
